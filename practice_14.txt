/* 7. Составьте список компаний, инвестиции которых оценены в долларах и их сумма более 1 миллиона.
(total_money_raised - строковое поле, поэтому используйте регулярное выражение $regex) */

[
  {
    $match:
      /**
       * query: The query in MQL.
       */
      {
        total_money_raised: {
          $ne: ""
        },
        total_money_raised: {
          $regex: /^\$\d+(\.\d+)?M$/
        }
      }
  },
  {
    $addFields:
      /**
       * newField: The new field name.
       * expression: The new field expression.
       */
      {
        total_mil_raised: {
          $toDouble: {
            $trim: {
              input: "$total_money_raised",
              chars: {
                $literal: "$M"
              }
            }
          }
        }
      }
  },
  {
    $match:
      /**
       * query: The query in MQL.
       */
      {
        total_mil_raised: {
          $gt: 1
        }
      }
  },
  {
    $count:
      /**
       * Provide the field name for the count.
       */
      "Количество компаний привлекших инвестиции в долларах на сумму более 1 миллиона"
  }
]
